// Microphone capture is local-first. Optional live recognition is separate from audio storage.
function createVoiceCapture(host) {
  const vEnv = host.environment || globalThis;
  const vDoc = vEnv.document;
  const vNow = () => vEnv.performance?.now?.() ?? Date.now();
  const vISO = () => new Date().toISOString();
  const vEscape = x => String(x ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
  const vIds = value => {try {const a = JSON.parse(value || '[]'); return Array.isArray(a) ? a.filter(x => typeof x === 'string') : [];} catch {return [];}};
  let vSheet, vPreviousFocus, vRecorder, vStream, vRecognition, vContext, vAnalyser, vFrame, vTimer, vTextTimer;
  let vState = 'Ready', vPaused = false, vBusy = false, vDraft = null, vError = '', vMode = 'quick', vLang = 'id-ID', vSpeechMode = 'local';
  let vChunks = [], vQueue = Promise.resolve(), vStarted = 0, vElapsed = 0, vAudioURL = '', vStopResolve = null, vStopPromise = null, vDisposed = false, vRequestCanceled = false;
  let vRecognitionRunning = false, vLocalSpeech = false, vBrowserConsented = false, vRecoveryMissing = false, vSpeechStatus = 'Transkrip opsional. Audio tetap tersimpan tanpa transkrip.', vFinalSpeech = '', vInterimSpeech = '';
  const vData = () => {const d = host.getData(); return d.data || d;};
  const vToast = text => host.toast?.(text);
  const vDuration = () => vElapsed + (vState === 'Recording' && !vPaused ? Math.max(0, vNow() - vStarted) : 0);
  const vClock = ms => `${String(Math.floor(ms / 60000)).padStart(2,'0')}:${String(Math.floor(ms / 1000) % 60).padStart(2,'0')}`;
  const vTime = fn => (vEnv.setTimeout || setTimeout)(fn, 250);
  async function vSave() {
    if (!vDraft) return false;
    vDraft.updatedAt = vISO();
    vDraft.durationMs = Math.round(vDuration());
    vDraft.voiceChunkIds = JSON.stringify(vChunks.map(c => c.id).filter(Boolean));
    vDraft.attachmentIds = JSON.stringify([...new Set([...vIds(vDraft.attachmentIds), ...vChunks.map(c => c.id).filter(Boolean)])]);
    try {const ok = await host.save(); if (ok === false) throw Error('Pencatatan belum tersimpan. Coba lagi sebelum menutup.'); return true;}
    catch (err) {vError = err.message || 'Penyimpanan gagal'; vDraft.localStatus = 'save-pending'; vUpdate(); return false;}
  }
  function vFreshDraft() {
    const id = vEnv.crypto?.randomUUID?.() || `voice-${Date.now()}-${Math.random().toString(36).slice(2)}`;
    return {id, recordKind:'voice', captureKind:'voice', title:'Voice note', notes:'', rawText:'', createdAt:vISO(), updatedAt:vISO(),
      voiceState:'recording', voiceChunkIds:'[]', attachmentIds:'[]', durationMs:0, audioMime:'',
      localStatus:'draft', transcriptionStatus:'pending', backupStatus:'pending', audioDraft:true, speechProvider:'none', speechLanguage:vLang};
  }
  function vSet(state) {vState = state; if(vDraft) vDraft.voiceState = state.toLowerCase(); vRender();}
  function vCleanup() {
    if(vTimer) (vEnv.clearInterval || clearInterval)(vTimer); vTimer = null;
    if(vFrame) vEnv.cancelAnimationFrame?.(vFrame); vFrame = null;
    vRecognitionRunning = false;
    if(vRecognition) {vRecognition.onend = null; try {vRecognition.abort();} catch {}} vRecognition = null;
    vStream?.getTracks?.().forEach(track => track.stop()); vStream = null;
    try {vContext?.close?.()?.catch?.(() => {});} catch {} vContext = null; vAnalyser = null;
  }
  async function vPersistChunk(chunk) {
    if(chunk.id) return true;
    try {
      const file = new Blob([chunk.blob], {type:chunk.blob.type || vRecorder?.mimeType || 'audio/webm'});
      // Blob accepts a local name for storage; no credentials or network are involved.
      file.name = `voice-${vDraft.id}-${String(chunk.order).padStart(4,'0')}.${file.type.includes('mp4') ? 'm4a' : 'webm'}`;
      const asset = await host.storage.putAttachment(file); if(!asset?.id) throw Error('Audio belum tersimpan');
      chunk.id = asset.id; vDraft.audioMime = file.type; vDraft.localStatus = 'local';
      const saved=await vSave();if(saved)chunk.blob=null;return saved;
    } catch(err) {vError = `Audio chunk masih di memori: ${err.message}. Jangan tutup; gunakan Coba simpan lagi atau Unduh audio.`; if(vDraft) vDraft.localStatus = 'audio-save-pending'; vUpdate(); return false;}
  }
  function vOnChunk(event) {
    if(!event.data?.size || !vDraft) return;
    const chunk = {blob:event.data, id:'', order:vChunks.length}; vChunks.push(chunk);
    vQueue = vQueue.catch(() => {}).then(() => vPersistChunk(chunk));
  }
  async function vSpeechStart() {
    const SR = vEnv.SpeechRecognition || vEnv.webkitSpeechRecognition;
    if(vSpeechMode === 'off' || !SR) {vSpeechStatus = 'Audio saja. Transkrip dapat ditulis saat review.'; vUpdate(); return;}
    try {
      const candidate = new SR(); vLocalSpeech = false;
      if(vSpeechMode === 'local') {
        if(!('processLocally' in candidate) || typeof SR.available !== 'function') {vSpeechStatus = 'Transkrip lokal belum tersedia di browser ini. Audio tetap direkam.'; vUpdate(); return;}
        const available = await SR.available({langs:[vLang], processLocally:true});
        if(available !== 'available') {vSpeechStatus = `Paket bahasa lokal belum siap (${available}). Audio tetap direkam; tidak dialihkan ke cloud.`; vUpdate(); return;}
        candidate.processLocally = true; vLocalSpeech = true;
      } else if(vSpeechMode === 'browser') {
        if(!vBrowserConsented) {vSpeechStatus = 'Izinkan layanan browser terlebih dahulu. Audio tetap lokal.'; vUpdate(); return;}
        if('processLocally' in candidate) candidate.processLocally = false;
      }
      if(vState !== 'Recording' || vPaused || vDisposed) return;
      candidate.lang = vLang; candidate.continuous = true; candidate.interimResults = true; vRecognition = candidate;
      candidate.onresult = event => {
        let interim = '';
        for(let i = event.resultIndex; i < event.results.length; i++) {
          const result = event.results[i], text = result[0]?.transcript || '';
          if(result.isFinal) vFinalSpeech += (vFinalSpeech ? ' ' : '') + text.trim(); else interim += text;
        }
        vInterimSpeech = interim; vDraft.rawText = vFinalSpeech; vDraft.notes = vFinalSpeech;
        vDraft.transcriptionStatus = vFinalSpeech ? 'draft' : 'pending';
        vQueue = vQueue.catch(() => {}).then(vSave); vUpdate();
      };

function crCsvParse(text){const rows=[],row=[];let v='',quoted=false;const s=String(text||'').replace(/^\uFEFF/,'');for(let i=0;i<s.length;i++){let c=s[i];if(quoted){if(c==='"'&&s[i+1]==='"'){v+='"';i++}else if(c==='"')quoted=false;else v+=c}else if(c==='"')quoted=true;else if(c===','){row.push(v);v=''}else if(c==='\n'||c==='\r'){if(c==='\r'&&s[i+1]==='\n')i++;row.push(v);v='';if(row.some(x=>String(x).trim()))rows.push([...row]);row.length=0}else v+=c}if(quoted)throw Error('Unclosed quoted field');row.push(v);if(row.some(x=>String(x).trim()))rows.push([...row]);if(rows.length<2)throw Error('CSV needs a header and at least one data row');const hd=rows.shift().map(x=>x.trim().toLowerCase());if(new Set(hd).size!==hd.length)throw Error('Duplicate column names');return{headers:hd,rows:rows.map(arr=>Object.fromEntries(hd.map((k,i)=>[k,(arr[i]||'').trim()])))}}
